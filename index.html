2. Connexion Callback Javascript

// Fonction globale appelée après validation du code SMS/Appel
function creditUserAccount(creditsToAdd, providerName = 'SMS/Audiotel') {
    if (!currentUser) return;

    currentUser.balance += creditsToAdd;
    currentUser.rechargesCount += 1;

    currentUser.history.unshift({
        type: 'Recharge ' + providerName,
        details: 'Code validé avec succès',
        amount: '+' + creditsToAdd + ' crédits',
        date: new Date().toLocaleTimeString()
    });

    saveUserData();
    updateUI();
}
